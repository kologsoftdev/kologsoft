cmake_minimum_required(VERSION 3.1...3.31)
project(runner LANGUAGES CXX)

cmake_policy(SET CMP0063 NEW)

set(CMAKE_INSTALL_RPATH "$ORIGIN/lib")

# ---------------------------------------------------------------------------
# Create the executable target
# ---------------------------------------------------------------------------
add_executable(${BINARY_NAME} WIN32)

# ---------------------------------------------------------------------------
# Include directories
# ---------------------------------------------------------------------------
target_include_directories(${BINARY_NAME} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/.."
)

# ---------------------------------------------------------------------------
# Source files
# ---------------------------------------------------------------------------
target_sources(${BINARY_NAME} PRIVATE
        "flutter_window.cpp"
        "main.cpp"
        "utils.cpp"
        "win32_window.cpp"
        "printer_channel.cpp"        # Kologsoft silent printing channel
        "../flutter/generated_plugin_registrant.cc"
)

# ---------------------------------------------------------------------------
# Link libraries
# winspool.lib  — Windows Print Spooler API (OpenPrinter, WritePrinter etc.)
# dwmapi.lib    — Desktop Window Manager API (DwmSetWindowAttribute)
# ---------------------------------------------------------------------------
target_link_libraries(${BINARY_NAME} PRIVATE
        flutter
        flutter_wrapper_app
        winspool.lib
        dwmapi.lib
)

# ---------------------------------------------------------------------------
# Standard Flutter runner compile options — keep as generated by Flutter
# ---------------------------------------------------------------------------
target_compile_definitions(${BINARY_NAME} PRIVATE
        "FLUTTER_VERSION=\"${FLUTTER_VERSION}\""
        "FLUTTER_VERSION_MAJOR=${FLUTTER_VERSION_MAJOR}"
        "FLUTTER_VERSION_MINOR=${FLUTTER_VERSION_MINOR}"
        "FLUTTER_VERSION_PATCH=${FLUTTER_VERSION_PATCH}"
        "FLUTTER_VERSION_BUILD=${FLUTTER_VERSION_BUILD}"
)

apply_standard_settings(${BINARY_NAME})
